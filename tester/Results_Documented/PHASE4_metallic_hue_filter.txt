"""
tune_params_phase5.py — Background-rejection filter + param2 fine-tune
=======================================================================

Phase 4 showed the metallic ACCEPTANCE filter rejects real coins (complex
engravings make interiors <20% "gold"-colored).

New strategy — background REJECTION:
  Instead of requiring the interior to look like a coin, we reject circles
  whose interior is a known non-coin background color:
    • Teal/cyan painted wood : H ∈ [70, 110],  S > 25   (fixes 011, 036)
    • Saturated red rubber   : H ∈ [0,  10] or [160,180], S > 100  (fixes 044, 134)

  These ranges cannot be confused with any euro coin (gold H≈15-28,
  silver S<50, copper H≈8-20).

Phase 4 also showed FN comes from coins not generating Hough votes at
param2=50. Try param2 ∈ [40, 45, 50] combined with the background-reject
filter and existing sharpness filter.

Sweep grid (18 combos, ~8 min):
  bg_reject_on : [True, False]
  bg_min       : [0.30, 0.40, 0.50]   (fraction of interior that must be NON-background)
  param2       : [40, 45, 50]

Fixed: sh_min=1.5, arc_min=0.35, ed_min=0.08, cu_min=0.25, br_min=30.0

Usage:
    $env:PYTHONPATH="."; python tester/tune_params_phase5.py
"""

MAX_IMAGE_DIM = 960
PARAM1     = 40
MIN_DIST   = 80
ARC_MIN    = 0.35
ED_MIN     = 0.08
CU_MIN     = 0.25
BR_MIN     = 30.0
NMS_THRESH = 0.5
RING_WIDTH = 8
SH_MIN     = 1.5   # fixed — best from Phase 3

# Sweep
BG_REJECT_ON_VALUES = [True, False]
BG_MIN_VALUES       = [0.30, 0.40, 0.50]
PARAM2_VALUES       = [40, 45, 50]

# ===========================================================================
# Precomputed sampling arrays
# ===========================================================================
_N72 = 72
_A72 = np.linspace(0, 2 * np.pi, _N72, endpoint=False)
_C72 = np.cos(_A72)
_S72 = np.sin(_A72)

_FRACS = np.linspace(0.15, 0.90, 8)
_N_DISC = 20
_Ad = np.linspace(0, 2 * np.pi, _N_DISC, endpoint=False)
_Cd, _Sd = np.cos(_Ad), np.sin(_Ad)

_N_SEC = 36
_A36 = np.linspace(0, 2 * np.pi, _N_SEC, endpoint=False)
_C36, _S36 = np.cos(_A36), np.sin(_A36)
_SEC_I = (_A36 / (2 * np.pi) * _N_SEC).astype(int) % _N_SEC

# ===========================================================================
# Filter helpers
# ===========================================================================

def _pts(cx, cy, radius, C, S, H, W):
    xs = np.clip(np.round(cx + radius * C).astype(np.int32), 0, W - 1)
    ys = np.clip(np.round(cy + radius * S).astype(np.int32), 0, H - 1)
    return xs, ys


def _nms(circles, thresh):
    circles = sorted(circles, key=lambda c: c[2], reverse=True)
    keep = []
    for x1, y1, r1 in circles:
        if all(np.hypot(x1-x2, y1-y2) >= (r1+r2)*(1-thresh) for x2,y2,r2 in keep):
            keep.append((x1, y1, r1))
    return keep


def _arc(edges, cx, cy, r):
    H, W = edges.shape
    hit = np.zeros(_N_SEC, dtype=bool)
    for frac in (1 - RING_WIDTH/max(r,1), 1.0, 1 + RING_WIDTH/max(r,1)):
        xs, ys = _pts(cx, cy, r*frac, _C36, _S36, H, W)
        hit[_SEC_I[edges[ys, xs] > 0]] = True
    return hit.sum() / _N_SEC


def _edge_density(edges, cx, cy, r):
    H, W = edges.shape
    ri, ro = max(r - RING_WIDTH, 1), r + RING_WIDTH
    hits = tot = 0
    for frac in np.linspace(ri/max(r,1), ro/max(r,1), 5):
        xs, ys = _pts(cx, cy, r*frac, _C72, _S72, H, W)
        hits += int((edges[ys, xs] > 0).sum())
        tot  += _N72
    return hits/tot if tot else 0.0


def _cu(hsv, cx, cy, r):
    H, W = hsv.shape[:2]
    inner_r = max(int(r*0.80), 1)
    hues, sats, vals = [], [], []
    for frac in _FRACS:
        xs, ys = _pts(cx, cy, inner_r*frac, _Cd, _Sd, H, W)
        p = hsv[ys, xs]
        hues.append(p[:,0]); sats.append(p[:,1]); vals.append(p[:,2])
    hue = np.concatenate(hues).astype(np.float32)
    sat = np.concatenate(sats)
    val = np.concatenate(vals).astype(np.float32)
    col = sat > 12
    if col.sum() >= 10:
        return float(1/(1 + np.std(hue[col])/30))
    return float(1/(1 + np.std(val)/40)) if len(val) else 0.0


def _brightness(hsv, cx, cy, r):
    H, W = hsv.shape[:2]
    inner_r = max(int(r*0.75), 1)
    vals = []
    for frac in _FRACS:
        xs, ys = _pts(cx, cy, inner_r*frac, _Cd, _Sd, H, W)
        vals.append(hsv[ys, xs, 2])
    v = np.concatenate(vals).astype(np.float32)
    return float(np.mean(v)) if len(v) else 0.0


def _edge_density_at_r(edges, cx, cy, sample_r):
    H, W = edges.shape
    xs = np.clip(np.round(cx + sample_r * _C72).astype(np.int32), 0, W - 1)
    ys = np.clip(np.round(cy + sample_r * _S72).astype(np.int32), 0, H - 1)
    return float((edges[ys, xs] > 0).mean())


def _ring_sharpness(edges, cx, cy, r):
    ring  = _edge_density_at_r(edges, cx, cy, r)
    inner = _edge_density_at_r(edges, cx, cy, max(r * 0.55, 1.0))
    outer = _edge_density_at_r(edges, cx, cy, r * 1.55)
    bg = (inner + outer) * 0.5
    if bg < 1e-4:
        return ring * 100.0
    return ring / bg


def _background_fraction(hsv, cx, cy, r):
    """
    Fraction of interior pixels that are a KNOWN NON-COIN background color.

    We do NOT check if the interior looks like a coin (Phase 4 lesson: coin
    engravings make interiors <20% 'gold'-colored even for real coins).

    Instead we flag two specific background signatures that appear in the
    problem images and are impossible to confuse with any euro coin:

    Teal / cyan painted wood  (011, 036, 057, 079):
        H ∈ [70, 110],  S > 25
        Teal is far outside every euro coin colour (gold H≈20, silver S<50,
        copper H≈13). A coin with any teal-coloured interior cannot be a coin.

    Saturated red / deep orange rubber  (044, 103, 134):
        (H ∈ [0, 10] OR H ∈ [160, 180])  AND  S > 100
        Rubber-red is highly saturated (S 130-200). Gold coins are warm
        but much lower saturation (S 50-120). Silver coins have S<50.

    Returns the fraction of pixels that ARE background (higher = more likely FP).
    If this fraction exceeds bg_min, the circle is rejected.
    """
    H_img, W_img = hsv.shape[:2]
    inner_r = max(int(r * 0.75), 1)
    mask = np.zeros((H_img, W_img), dtype=np.uint8)
    cv2.circle(mask, (cx, cy), inner_r, 255, -1)

    h = hsv[:, :, 0][mask > 0].astype(np.float32)
    s = hsv[:, :, 1][mask > 0].astype(np.float32)

    if len(h) == 0:
        return 0.0

    # Teal / cyan painted wood
    teal = ((h >= 70) & (h <= 110)) & (s > 25)

    # Saturated red rubber (OpenCV H: red wraps around 0/180)
    red_rubber = ((h <= 10) | (h >= 160)) & (s > 100)

    background = teal | red_rubber
    return float(background.mean())

# Preprocessing + cache

def _prep(img):
    h, s, v = cv2.split(cv2.cvtColor(img, cv2.COLOR_BGR2HSV))
    v   = cv2.createCLAHE(2.0, (8, 8)).apply(v)
    img = cv2.cvtColor(cv2.merge((h, s, v)), cv2.COLOR_HSV2BGR)
    gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
    sm   = cv2.GaussianBlur(cv2.bilateralFilter(gray, 9, 75, 75), (7, 7), 2.0)
    bl   = cv2.GaussianBlur(cv2.bilateralFilter(gray, 13, 90, 90), (9, 9), 2.5)
    e    = cv2.Canny(bl, 15, 50)
    k1   = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (3, 3))
    k2   = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (7, 7))
    edges = cv2.morphologyEx(cv2.dilate(e, k1, iterations=2),
                             cv2.MORPH_CLOSE, k2, iterations=2)
    hsv  = cv2.cvtColor(img, cv2.COLOR_BGR2HSV)
    return sm, edges, hsv


def build_cache(images_dir, gt):
    cache = []
    for fname, truth in gt.items():
        path = images_dir / fname
        if not path.exists(): print(f"  [WARN] {fname} not found"); continue
        img = cv2.imread(str(path))
        if img is None: print(f"  [WARN] cannot read {fname}"); continue
        h, w = img.shape[:2]
        sc = min(1.0, MAX_IMAGE_DIM / max(h, w))
        if sc < 1.0:
            img = cv2.resize(img, (int(w*sc), int(h*sc)), interpolation=cv2.INTER_AREA)
        sm, edges, hsv = _prep(img)
        cache.append((fname, sm, edges, hsv, sc, truth))
        print(f"  {fname}  {img.shape[1]}x{img.shape[0]}")
    return cache


# ===========================================================================
# Evaluate
# ===========================================================================

def evaluate(bg_reject_on, bg_min, param2, cache):
    total_err = total_over = 0
    per_image = {}

    for fname, sm, edges, hsv, sc, truth in cache:
        min_r = max(5,  int(30  * sc))
        max_r = max(10, int(220 * sc))
        min_d = max(10, int(MIN_DIST * sc))

        raw = cv2.HoughCircles(sm, cv2.HOUGH_GRADIENT,
                               dp=1, minDist=min_d,
                               param1=PARAM1, param2=param2,
                               minRadius=min_r, maxRadius=max_r)
        circles = [] if raw is None else np.uint16(np.around(raw[0])).tolist()
        circles = _nms(circles, NMS_THRESH)

        valid = []
        for x, y, r in circles:
            x, y, r = int(x), int(y), int(r)
            if _arc(edges, x, y, r)           < ARC_MIN:  continue
            if _edge_density(edges, x, y, r)  < ED_MIN:   continue
            if _cu(hsv, x, y, r)              < CU_MIN:   continue
            if _brightness(hsv, x, y, r)      < BR_MIN:   continue
            if _ring_sharpness(edges, x, y, r) < SH_MIN:  continue
            if bg_reject_on and _background_fraction(hsv, x, y, r) >= bg_min:
                continue
            valid.append((x, y, r))

        det = len(valid)
        per_image[fname] = det
        total_err  += abs(det - truth)
        total_over += max(0, det - truth)

    return total_err, total_over, per_image

    print(f"RANK-1  bg_reject={best_bg}  bg_min={best_bm}  param2={best_p2}")
