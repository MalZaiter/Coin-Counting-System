_N_SEC  = 36
_A36    = np.linspace(0, 2 * np.pi, _N_SEC, endpoint=False)
_C36    = np.cos(_A36)
_S36    = np.sin(_A36)
_SEC_I  = (_A36 / (2 * np.pi) * _N_SEC).astype(int) % _N_SEC

_N_RING = 72
_A72    = np.linspace(0, 2 * np.pi, _N_RING, endpoint=False)
_C72    = np.cos(_A72)
_S72    = np.sin(_A72)

_FRACS  = np.linspace(0.15, 0.90, 8)
_N_DISC = 20
_Ad     = np.linspace(0, 2 * np.pi, _N_DISC, endpoint=False)
_Cd     = np.cos(_Ad)
_Sd     = np.sin(_Ad)

# Helpers

def _pts(cx, cy, radius, C, S, H, W):
    xs = np.clip(np.round(cx + radius * C).astype(np.int32), 0, W - 1)
    ys = np.clip(np.round(cy + radius * S).astype(np.int32), 0, H - 1)
    return xs, ys


def _nms(circles, thresh):
    circles = sorted(circles, key=lambda c: c[2], reverse=True)
    keep = []
    for x1, y1, r1 in circles:
        if all(np.hypot(x1 - x2, y1 - y2) >= (r1 + r2) * (1 - thresh)
               for x2, y2, r2 in keep):
            keep.append((x1, y1, r1))
    return keep


def _arc(edges, cx, cy, r):
    H, W = edges.shape
    w    = RING_WIDTH
    hit  = np.zeros(_N_SEC, dtype=bool)
    for frac in (1 - w / max(r, 1), 1.0, 1 + w / max(r, 1)):
        xs, ys = _pts(cx, cy, r * frac, _C36, _S36, H, W)
        hit[_SEC_I[edges[ys, xs] > 0]] = True
    return hit.sum() / _N_SEC


def _edge_density_at_radius(edges, cx, cy, sample_r):
    """
    Mean edge density sampled uniformly around a single radius sample_r.
    Used by ring_edge_sharpness to probe at/inside/outside the coin boundary.
    """
    H, W  = edges.shape
    xs, ys = _pts(cx, cy, sample_r, _C72, _S72, H, W)
    return float((edges[ys, xs] > 0).mean())


def _ring_edge_sharpness(edges, cx, cy, r):
    """
    Ratio of edge density AT the coin boundary vs. background regions.

    Samples at three zones:
      ring   = at r              (the coin boundary)
      inner  = at 0.55 * r      (well inside the coin)
      outer  = at 1.55 * r      (outside the coin, background)

    sharpness = ring / mean(inner, outer)

    Real coin:
      Strong, clean edge at r → ring_density >> background → sharpness >> 1
    Textured background false positive:
      Edges uniform everywhere  → ring_density ≈ background  → sharpness ≈ 1

    Returns 0.0 when background density is essentially zero (avoids div/0).
    """
    ring  = _edge_density_at_radius(edges, cx, cy, r)
    inner = _edge_density_at_radius(edges, cx, cy, max(r * 0.55, 1))
    outer = _edge_density_at_radius(edges, cx, cy, r * 1.55)
    bg    = (inner + outer) * 0.5
    if bg < 1e-4:
        return ring * 100.0   # near-zero background → effectively infinite sharpness
    return ring / bg


def _edge_density(edges, cx, cy, r):
    H, W   = edges.shape
    ri, ro = max(r - RING_WIDTH, 1), r + RING_WIDTH
    hits = tot = 0
    for frac in np.linspace(ri / max(r, 1), ro / max(r, 1), 5):
        xs, ys = _pts(cx, cy, r * frac, _C72, _S72, H, W)
        hits  += int((edges[ys, xs] > 0).sum())
        tot   += _N_RING
    return hits / tot if tot else 0.0


def _cu(hsv, cx, cy, r):
    H, W    = hsv.shape[:2]
    inner_r = max(int(r * 0.80), 1)
    hues, sats, vals = [], [], []
    for frac in _FRACS:
        xs, ys = _pts(cx, cy, inner_r * frac, _Cd, _Sd, H, W)
        p = hsv[ys, xs]
        hues.append(p[:, 0])
        sats.append(p[:, 1])
        vals.append(p[:, 2])
    hue = np.concatenate(hues).astype(np.float32)
    sat = np.concatenate(sats)
    val = np.concatenate(vals).astype(np.float32)
    col = sat > 12
    if col.sum() >= 10:
        return float(1 / (1 + np.std(hue[col]) / 30))
    return float(1 / (1 + np.std(val) / 40)) if len(val) else 0.0


def _brightness(hsv, cx, cy, r):
    H, W    = hsv.shape[:2]
    inner_r = max(int(r * 0.75), 1)
    vals = []
    for frac in _FRACS:
        xs, ys = _pts(cx, cy, inner_r * frac, _Cd, _Sd, H, W)
        vals.append(hsv[ys, xs, 2])
    v = np.concatenate(vals).astype(np.float32)
    return float(np.mean(v)) if len(v) else 0.0

def _prep(img):
    h, s, v = cv2.split(cv2.cvtColor(img, cv2.COLOR_BGR2HSV))
    v    = cv2.createCLAHE(2.0, (8, 8)).apply(v)
    img  = cv2.cvtColor(cv2.merge((h, s, v)), cv2.COLOR_HSV2BGR)
    gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)

    # Smoothed grayscale for HoughCircles
    sm = cv2.GaussianBlur(cv2.bilateralFilter(gray, 9, 75, 75), (7, 7), 2.0)

    # Enhanced edge map for arc/density filters
    bl     = cv2.GaussianBlur(cv2.bilateralFilter(gray, 13, 90, 90), (9, 9), 2.5)
    e      = cv2.Canny(bl, 15, 50)
    k1     = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (3, 3))
    k2     = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (7, 7))
    edges  = cv2.morphologyEx(cv2.dilate(e, k1, iterations=2),
                              cv2.MORPH_CLOSE, k2, iterations=2)

    hsv = cv2.cvtColor(img, cv2.COLOR_BGR2HSV)
    return sm, edges, hsv


def build_cache(images_dir, gt):
    cache = []
    for fname, truth in gt.items():
        path = images_dir / fname
        if not path.exists():
            print(f"  [WARN] {fname} not found")
            continue
        img = cv2.imread(str(path))
        if img is None:
            print(f"  [WARN] cannot read {fname}")
            continue
        h, w = img.shape[:2]
        sc = min(1.0, MAX_IMAGE_DIM / max(h, w))
        if sc < 1.0:
            img = cv2.resize(img, (int(w * sc), int(h * sc)),
                             interpolation=cv2.INTER_AREA)
        sm, edges, hsv = _prep(img)
        cache.append((fname, sm, edges, hsv, sc, truth))
        print(f"  {fname}  {img.shape[1]}×{img.shape[0]}")
    return cache


# ===========================================================================
# Evaluate one (param2, sharpness_min) combination
# ===========================================================================

def evaluate(p2, sharpness_min, cache):
    total_err = total_over = 0
    per_image = {}

    for fname, sm, edges, hsv, sc, truth in cache:
        min_r = max(5,  int(30  * sc))
        max_r = max(10, int(220 * sc))
        min_d = max(10, int(MIN_DIST * sc))

        raw = cv2.HoughCircles(sm, cv2.HOUGH_GRADIENT,
                               dp=1, minDist=min_d,
                               param1=PARAM1, param2=p2,
                               minRadius=min_r, maxRadius=max_r)

        circles = ([] if raw is None
                   else np.uint16(np.around(raw[0])).tolist())
        circles = _nms(circles, NMS_THRESH)

        valid = []
        for x, y, r in circles:
            x, y, r = int(x), int(y), int(r)

            # --- existing filters (Phase 1 optimised) ---
            if _arc(edges, x, y, r)            < ARC_MIN: continue
            if _edge_density(edges, x, y, r)   < ED_MIN:  continue
            if _cu(hsv, x, y, r)               < CU_MIN:  continue
            if _brightness(hsv, x, y, r)       < BR_MIN:  continue

            # --- Phase 3: ring edge sharpness filter ---
            if sharpness_min > 0.0:
                if _ring_edge_sharpness(edges, x, y, r) < sharpness_min:
                    continue

            valid.append((x, y, r))

        det = len(valid)
        per_image[fname]  = det
        total_err        += abs(det - truth)
        total_over       += max(0, det - truth)

    return total_err, total_over, per_image

    print(f"sharpness_min   : {SHARPNESS_MIN_VALUES}")
    print(f"Combinations    : {len(PARAM2_VALUES) * len(SHARPNESS_MIN_VALUES)}")

    if not IMAGES_DIR.exists():
        print(f"\n[ERROR] IMAGES_DIR not found: {IMAGES_DIR.resolve()}")
        sys.exit(1)

    print(f"\nCaching {len(GROUND_TRUTH)} images…")
    t0    = time.time()
    cache = build_cache(IMAGES_DIR, GROUND_TRUTH)
    n_img = len(cache)
    print(f"\n  {n_img} images ready in {time.time() - t0:.1f}s\n")

    combos = list(itertools.product(PARAM2_VALUES, SHARPNESS_MIN_VALUES))
    total  = len(combos)

    results = []
    t0 = time.time()
    for i, (p2, sh) in enumerate(combos):
        err, over, pi = evaluate(p2, sh, cache)
        results.append((err, over, p2, sh, pi))
        elapsed = time.time() - t0
        eta     = elapsed / (i + 1) * (total - i - 1)
        label   = f"p2={p2:3d}  sh_min={sh:.2f}"
        print(f"  {label}  error={err:4d}  over={over:4d}  "
              f"ETA {eta / 60:.1f}m", flush=True)

    results.sort(key=lambda r: (r[0], r[1]))

    print(f"\n{'=' * 65}")
    print(f"TOP {TOP_N} RESULTS")
    print(f"{'=' * 65}")
    print(f"  {'param2':>7}  {'sh_min':>7}  {'TotalErr':>8}  "
          f"{'OverCnt':>7}  {'UnderCnt':>9}")
    print(f"  {'-' * 50}")
    for err, over, p2, sh, _ in results[:TOP_N]:
        under  = err - over
        marker = " ← BEST" if (err, over, p2, sh) == results[0][:4] else ""
        print(f"  {p2:7d}  {sh:7.2f}  {err:8d}  {over:7d}  {under:9d}{marker}")

    best_err, best_over, best_p2, best_sh, best_pi = results[0]
    best_under = best_err - best_over

    print(f"\n{'=' * 65}")
    print(f"RANK-1  param2={best_p2}  sharpness_min={best_sh}"
          f"  error={best_err}  over={best_over}  under={best_under}")
    print(f"{'=' * 65}")

    n_exact = 0
    worst   = []
    for fname, truth in GROUND_TRUTH.items():
        if fname not in best_pi:
            continue
        det  = best_pi[fname]
        diff = det - truth
        flag = "✓" if diff == 0 else ("↑" if diff > 0 else "↓")
        print(f"  {flag} {fname:<12}  det={det:2d}  truth={truth}  {diff:+d}")
        if diff == 0:
            n_exact += 1
        else:
            worst.append((fname, det, truth, diff))

    acc = 100 * n_exact / n_img if n_img else 0
    print(f"\n  Exact: {n_exact}/{n_img}  ({acc:.0f}%)")

    if worst:
        print(f"\n  Remaining misses:")
        for fname, det, truth, diff in sorted(worst, key=lambda x: -abs(x[3])):
            print(f"    {fname}  det={det}  truth={truth}  {diff:+d}")

    # -----------------------------------------------------------------------
    # Show per-param2 summary (helps decide if raising param2 alone is enough)
    # -----------------------------------------------------------------------
    print(f"\n{'=' * 65}")
    print("PER-param2 BEST  (best sharpness_min for each param2 value)")
    print(f"{'=' * 65}")
    print(f"  {'param2':>7}  {'best_sh':>8}  {'TotalErr':>8}  "
          f"{'OverCnt':>7}  {'Exact':>7}")
    for p2 in PARAM2_VALUES:
        sub = [(err, over, sh, pi)
               for err, over, pp, sh, pi in results if pp == p2]
        sub.sort(key=lambda x: (x[0], x[1]))
        if not sub:
            continue
        err, over, sh, pi = sub[0]
        n_ex = sum(1 for fn, tr in GROUND_TRUTH.items()
                   if fn in pi and pi[fn] == tr)
        print(f"  {p2:7d}  {sh:8.2f}  {err:8d}  {over:7d}  "
              f"{n_ex:4d}/{n_img} ({100*n_ex/n_img:.0f}%)")

    print(f"\n{'=' * 65}")
    print("PASTE INTO detect.py / detect_circles()")
    print(f"{'=' * 65}")
    print(f"""
circles = cv2.HoughCircles(
    smoothed, cv2.HOUGH_GRADIENT,
    dp=1,
    minDist={MIN_DIST},
    param1={PARAM1},
    param2={best_p2},          # Phase 3 optimal
    minRadius=30,
    maxRadius=220,
)

# filter_false_positives() — add ring_sharpness_min={best_sh}
""")


if __name__ == "__main__":
    main()
