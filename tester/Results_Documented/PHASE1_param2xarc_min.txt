_N_SEC  = 36

_C36    = np.cos(_A36);  _S36 = np.sin(_A36)
_SEC_I  = (_A36 / (2*np.pi) * _N_SEC).astype(int) % _N_SEC

# ===========================================================================
# Filter helpers

def _pts(cx, cy, radius, C, S, H, W):
    xs = np.clip(np.round(cx + radius * C).astype(np.int32), 0, W-1)
    ys = np.clip(np.round(cy + radius * S).astype(np.int32), 0, H-1)
    return xs, ys


def _nms(circles, thresh):
    circles = sorted(circles, key=lambda c: c[2], reverse=True)
    keep = []
    for x1, y1, r1 in circles:
        if all(np.hypot(x1-x2, y1-y2) >= (r1+r2)*(1-thresh) for x2,y2,r2 in keep):
            keep.append((x1, y1, r1))
    return keep


def _arc(edges, cx, cy, r):
    H, W = edges.shape
    w    = FIXED["ring_width"]
    hit  = np.zeros(_N_SEC, dtype=bool)
    for frac in (1 - w/max(r,1), 1.0, 1 + w/max(r,1)):
        xs, ys = _pts(cx, cy, r*frac, _C36, _S36, H, W)
        hit[_SEC_I[edges[ys, xs] > 0]] = True
    return hit.sum() / _N_SEC


def _edge_density(edges, cx, cy, r):
    H, W  = edges.shape
    w     = FIXED["ring_width"]
    ri, ro = max(r-w, 1), r+w
    hits = tot = 0
    for frac in np.linspace(ri/max(r,1), ro/max(r,1), 5):
        xs, ys = _pts(cx, cy, r*frac, _C72, _S72, H, W)
        hits  += int((edges[ys, xs] > 0).sum())
        tot   += _N_RING
    return hits/tot if tot else 0.0


def _cu(hsv, cx, cy, r):
    H, W    = hsv.shape[:2]
    inner_r = max(int(r*0.80), 1)
    hues, sats, vals = [], [], []
    for frac in _FRACS:
        xs, ys = _pts(cx, cy, inner_r*frac, _Cd, _Sd, H, W)
        p = hsv[ys, xs]
        hues.append(p[:,0]); sats.append(p[:,1]); vals.append(p[:,2])
    hue = np.concatenate(hues).astype(np.float32)
    sat = np.concatenate(sats)
    val = np.concatenate(vals).astype(np.float32)
    col = sat > 12
    if col.sum() >= 10:
        return float(1/(1 + np.std(hue[col])/30))
    return float(1/(1 + np.std(val)/40)) if len(val) else 0.0


def _brightness(hsv, cx, cy, r):
    H, W    = hsv.shape[:2]
    inner_r = max(int(r*0.75), 1)
    vals = []
    for frac in _FRACS:
        xs, ys = _pts(cx, cy, inner_r*frac, _Cd, _Sd, H, W)
        vals.append(hsv[ys, xs, 2])
    v = np.concatenate(vals).astype(np.float32)
    return float(np.mean(v)) if len(v) else 0.0

def _prep(img):
    enh  = cv2.cvtColor(img, cv2.COLOR_BGR2HSV)
    h,s,v = cv2.split(enh)
    v     = cv2.createCLAHE(2.0,(8,8)).apply(v)
    img   = cv2.cvtColor(cv2.merge((h,s,v)), cv2.COLOR_HSV2BGR)
    gray  = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
    sm    = cv2.GaussianBlur(cv2.bilateralFilter(gray,9,75,75),(7,7),2.0)
    bl    = cv2.GaussianBlur(cv2.bilateralFilter(gray,13,90,90),(9,9),2.5)
    e     = cv2.Canny(bl,15,50)
    k1    = cv2.getStructuringElement(cv2.MORPH_ELLIPSE,(3,3))
    k2    = cv2.getStructuringElement(cv2.MORPH_ELLIPSE,(7,7))
    edges = cv2.morphologyEx(cv2.dilate(e,k1,iterations=2),cv2.MORPH_CLOSE,k2,iterations=2)
    hsv   = cv2.cvtColor(img, cv2.COLOR_BGR2HSV)
    return sm, edges, hsv


def build_cache(images_dir, gt):
    cache = []
    for fname, truth in gt.items():
        path = images_dir / fname
        if not path.exists(): print(f"  [WARN] {fname} not found"); continue
        img = cv2.imread(str(path))
        if img is None: print(f"  [WARN] cannot read {fname}"); continue
        h,w = img.shape[:2]
        sc  = min(1.0, MAX_IMAGE_DIM/max(h,w))
        if sc < 1.0:
            img = cv2.resize(img,(int(w*sc),int(h*sc)),interpolation=cv2.INTER_AREA)
        sm, edges, hsv = _prep(img)
        cache.append((fname, sm, edges, hsv, sc, truth))
        print(f"  {fname}  {img.shape[1]}×{img.shape[0]}")
    return cache


# ===========================================================================
# Evaluate
# ===========================================================================

def evaluate(p2, arc_min, cache):
    total_err = total_over = 0
    per_image = {}
    for fname, sm, edges, hsv, sc, truth in cache:
        min_r = max(5,  int(FIXED["min_radius"]*sc))
        max_r = max(10, int(FIXED["max_radius"]*sc))
        min_d = max(10, int(FIXED["min_dist"]*sc))
        raw   = cv2.HoughCircles(sm, cv2.HOUGH_GRADIENT,
                    dp=1, minDist=min_d,
                    param1=FIXED["hough_param1"], param2=p2,
                    minRadius=min_r, maxRadius=max_r)
        circles = [] if raw is None else np.uint16(np.around(raw[0])).tolist()
        circles = _nms(circles, FIXED["nms_thresh"])
        valid = []
        for x,y,r in circles:
            x,y,r = int(x),int(y),int(r)
            if _arc(edges,x,y,r)      < arc_min:          continue
            if _edge_density(edges,x,y,r) < FIXED["ed_min"]: continue
            if _cu(hsv,x,y,r)         < FIXED["cu_min"]:  continue
            if _brightness(hsv,x,y,r) < FIXED["br_min"]:  continue
            valid.append((x,y,r))
        det = len(valid)
        per_image[fname]  = det
        total_err         += abs(det-truth)
        total_over        += max(0,det-truth)
    return total_err, total_over, per_image

def main():
    print("=" * 60)
    print("Coin Detection — Quick 2-Parameter Sweep")
    print("=" * 60)

    if not IMAGES_DIR.exists():
        print(f"\n[ERROR] IMAGES_DIR not found: {IMAGES_DIR.resolve()}")
        sys.exit(1)

    print(f"\nCaching {len(GROUND_TRUTH)} images…")
    t0    = time.time()
    cache = build_cache(IMAGES_DIR, GROUND_TRUTH)
    print(f"\n  {len(cache)} images ready in {time.time()-t0:.1f}s\n")

    combos = list(itertools.product(PARAM2_VALUES, ARC_MIN_VALUES))
    total  = len(combos)
    print(f"Sweep: {len(PARAM2_VALUES)} param2 × {len(ARC_MIN_VALUES)} arc_min "
          f"= {total} combos\n")

    results = []
    t0 = time.time()
    for i, (p2, arc) in enumerate(combos):
        err, over, pi = evaluate(p2, arc, cache)
        results.append((err, over, p2, arc, pi))
        elapsed = time.time()-t0
        eta     = elapsed/(i+1)*(total-i-1)
        print(f"  param2={p2:2d}  arc_min={arc:.2f}  "
              f"error={err:4d}  over={over:3d}  "
              f"ETA {eta/60:.1f}m", flush=True)

    results.sort(key=lambda r: (r[0], r[1]))

    print(f"\n{'='*60}")
    print(f"TOP {TOP_N} RESULTS")
    print(f"{'='*60}")
    print(f"  {'param2':>7}  {'arc_min':>7}  {'TotalErr':>8}  {'OverCnt':>7}")
    print(f"  {'-'*40}")
    for err, over, p2, arc, _ in results[:TOP_N]:
        marker = " ← BEST" if (err,over,p2,arc) == results[0][:4] else ""
        print(f"  {p2:7d}  {arc:7.2f}  {err:8d}  {over:7d}{marker}")

    best_err, best_over, best_p2, best_arc, best_pi = results[0]

    print(f"\n{'='*60}")
    print(f"RANK-1  param2={best_p2}  arc_min={best_arc}"
          f"  error={best_err}  overcounts={best_over}")
    print(f"{'='*60}")

    n_imgs = n_exact = 0
    worst = []
    for fname, truth in GROUND_TRUTH.items():
        if fname not in best_pi: continue
        det  = best_pi[fname]
        diff = det - truth
        flag = "✓" if diff == 0 else ("↑" if diff > 0 else "↓")
        print(f"  {flag} {fname:<12}  det={det}  truth={truth}  {diff:+d}")
        n_imgs += 1
        if diff == 0: n_exact += 1
        else: worst.append((fname, det, truth, diff))

    acc = 100*n_exact/n_imgs if n_imgs else 0
    print(f"\n  Exact: {n_exact}/{n_imgs}  ({acc:.0f}%)")

    if worst:
        print(f"\n  Biggest misses:")
        for fname, det, truth, diff in sorted(worst, key=lambda x: -abs(x[3])):
            if abs(diff) >= 2:
                print(f"    {fname}  det={det}  truth={truth}  {diff:+d}")

    print(f"\n{'='*60}")
    print("PASTE INTO detect.py")
    print("=" * 60)
    print(f"""

    smoothed, cv2.HOUGH_GRADIENT,

    param2={best_p2},

    maxRadius=220,
)
# filter_false_positives thresholds:
arc_coverage_min    = {best_arc}
edge_density_min    = {FIXED['ed_min']}
color_uniformity_min = {FIXED['cu_min']}
brightness_min      = {FIXED['br_min']}
""")


if __name__ == "__main__":
