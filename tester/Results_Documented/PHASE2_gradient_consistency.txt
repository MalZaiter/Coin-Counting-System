================================================================================
ANALYSIS — Why gc_min failed
================================================================================

The gradient consistency filter is mathematically limited on uniform textures.

For a surface with gradients pointing in ALL directions equally (fabric weave,
wood grain, etc.), the expected value of mean |cos(theta)| is:

    E[|cos(theta)|] = 2/pi ≈ 0.637

This means ANY isotropic texture will score ~0.637 on the GC metric, which
is ABOVE any threshold that would also keep real coins (which score 0.45-0.75).
You cannot set gc_min high enough to reject these textures without also
rejecting real coins. The filter has a mathematical ceiling.

Result: even gc_min=0.50 only reduces error by 3 (from 210 to 207).
The same 15 problem images remain completely unaffected.

CONCLUSION: gradient_consistency is not the right tool for this problem.
Proceed to Phase 3 — ring edge sharpness filter.

_N_SEC = 36

_SEC_I = (_A36 / (2*np.pi) * _N_SEC).astype(int) % _N_SEC

# Filter helpers (fast angle-based sampling)

def _pts(cx, cy, radius, C, S, H, W):
    xs = np.clip(np.round(cx + radius*C).astype(np.int32), 0, W-1)
    ys = np.clip(np.round(cy + radius*S).astype(np.int32), 0, H-1)
    return xs, ys


def _nms(circles, thresh):
    circles = sorted(circles, key=lambda c: c[2], reverse=True)
    keep = []
    for x1, y1, r1 in circles:
        if all(np.hypot(x1-x2, y1-y2) >= (r1+r2)*(1-thresh) for x2,y2,r2 in keep):
            keep.append((x1, y1, r1))
    return keep


def _arc(edges, cx, cy, r):
    H, W = edges.shape
    w = RING_WIDTH
    hit = np.zeros(_N_SEC, dtype=bool)
    for frac in (1-w/max(r,1), 1.0, 1+w/max(r,1)):
        xs, ys = _pts(cx, cy, r*frac, _C36, _S36, H, W)
        hit[_SEC_I[edges[ys, xs] > 0]] = True
    return hit.sum() / _N_SEC


def _edge_density(edges, cx, cy, r):
    H, W = edges.shape
    w = RING_WIDTH
    ri, ro = max(r-w, 1), r+w
    hits = tot = 0
    for frac in np.linspace(ri/max(r,1), ro/max(r,1), 5):
        xs, ys = _pts(cx, cy, r*frac, _C72, _S72, H, W)
        hits += int((edges[ys, xs] > 0).sum())
        tot  += _N_RING
    return hits/tot if tot else 0.0


def _cu(hsv, cx, cy, r):
    H, W = hsv.shape[:2]
    inner_r = max(int(r*0.80), 1)
    hues, sats, vals = [], [], []
    for frac in _FRACS:
        xs, ys = _pts(cx, cy, inner_r*frac, _Cd, _Sd, H, W)
        p = hsv[ys, xs]
        hues.append(p[:,0]); sats.append(p[:,1]); vals.append(p[:,2])
    hue = np.concatenate(hues).astype(np.float32)
    sat = np.concatenate(sats)
    val = np.concatenate(vals).astype(np.float32)
    col = sat > 12
    if col.sum() >= 10:
        return float(1/(1 + np.std(hue[col])/30))
    return float(1/(1 + np.std(val)/40)) if len(val) else 0.0


def _brightness(hsv, cx, cy, r):
    H, W = hsv.shape[:2]
    inner_r = max(int(r*0.75), 1)
    vals = []
    for frac in _FRACS:
        xs, ys = _pts(cx, cy, inner_r*frac, _Cd, _Sd, H, W)
        vals.append(hsv[ys, xs, 2])
    v = np.concatenate(vals).astype(np.float32)
    return float(np.mean(v)) if len(v) else 0.0


def _gradient_consistency(gray, cx, cy, r, n=36):
    """
    Mean |cos(theta)| between Sobel gradient direction and outward radial direction.

    Real coin edges:     0.45-0.75  (gradients point radially)
    Textured background: 0.15-0.35  (gradients point randomly)

    NOTE: This filter is fundamentally limited. For any isotropic texture,
    the expected score is 2/pi = 0.637, which overlaps with real coin scores.
    See Phase 2 analysis for details.
    """
    H, W  = gray.shape
    r_out = r + RING_WIDTH
    y0 = max(0, cy - r_out);  y1 = min(H, cy + r_out + 1)
    x0 = max(0, cx - r_out);  x1 = min(W, cx + r_out + 1)

    crop = gray[y0:y1, x0:x1]
    gx   = cv2.Sobel(crop, cv2.CV_32F, 1, 0, ksize=3)
    gy   = cv2.Sobel(crop, cv2.CV_32F, 0, 1, ksize=3)

    angles = np.linspace(0, 2*np.pi, n, endpoint=False)
    cos_a  = np.cos(angles)
    sin_a  = np.sin(angles)

    scores = []
    for frac in (0.9, 1.0, 1.1):
        sr = r * frac
        xi = np.clip(np.round(cx + sr*cos_a).astype(int), x0, x1-1) - x0
        yi = np.clip(np.round(cy + sr*sin_a).astype(int), y0, y1-1) - y0
        gx_s = gx[yi, xi];  gy_s = gy[yi, xi]
        mag  = np.hypot(gx_s, gy_s)
        strong = mag > 5.0
        if strong.sum() < 3:
            continue
        dot = np.abs(gx_s[strong]*cos_a[strong] + gy_s[strong]*sin_a[strong])
        scores.extend((dot / mag[strong]).tolist())

    return float(np.mean(scores)) if scores else 0.0

def _prep(img):
    h, s, v = cv2.split(cv2.cvtColor(img, cv2.COLOR_BGR2HSV))
    v   = cv2.createCLAHE(2.0, (8,8)).apply(v)
    img = cv2.cvtColor(cv2.merge((h,s,v)), cv2.COLOR_HSV2BGR)
    gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
    sm   = cv2.GaussianBlur(cv2.bilateralFilter(gray,9,75,75),(7,7),2.0)
    bl   = cv2.GaussianBlur(cv2.bilateralFilter(gray,13,90,90),(9,9),2.5)
    e    = cv2.Canny(bl, 15, 50)
    k1   = cv2.getStructuringElement(cv2.MORPH_ELLIPSE,(3,3))
    k2   = cv2.getStructuringElement(cv2.MORPH_ELLIPSE,(7,7))
    edges = cv2.morphologyEx(cv2.dilate(e,k1,iterations=2),
                             cv2.MORPH_CLOSE,k2,iterations=2)
    hsv   = cv2.cvtColor(img, cv2.COLOR_BGR2HSV)
    return sm, edges, hsv, gray          # gray needed for gradient filter


def build_cache(images_dir, gt):
    cache = []
    for fname, truth in gt.items():
        path = images_dir / fname
        if not path.exists(): print(f"  [WARN] {fname} not found"); continue
        img = cv2.imread(str(path))
        if img is None: print(f"  [WARN] cannot read {fname}"); continue
        h, w = img.shape[:2]
        sc = min(1.0, MAX_IMAGE_DIM / max(h, w))
        if sc < 1.0:
            img = cv2.resize(img,(int(w*sc),int(h*sc)),interpolation=cv2.INTER_AREA)
        sm, edges, hsv, gray = _prep(img)
        cache.append((fname, sm, edges, hsv, gray, sc, truth))
        print(f"  {fname}  {img.shape[1]}x{img.shape[0]}")
    return cache


# ===========================================================================
# Evaluate one gc_min value
# ===========================================================================

def evaluate(gc_min, cache):
    total_err = total_over = 0
    per_image = {}
    for fname, sm, edges, hsv, gray, sc, truth in cache:
        min_r = max(5,  int(30  * sc))
        max_r = max(10, int(220 * sc))
        min_d = max(10, int(MIN_DIST * sc))
        raw   = cv2.HoughCircles(sm, cv2.HOUGH_GRADIENT,
                    dp=1, minDist=min_d,
                    param1=PARAM1, param2=PARAM2,
                    minRadius=min_r, maxRadius=max_r)
        circles = [] if raw is None else np.uint16(np.around(raw[0])).tolist()
        circles = _nms(circles, NMS_THRESH)
        valid = []
        for x, y, r in circles:
            x, y, r = int(x), int(y), int(r)
            if _arc(edges,x,y,r)          < ARC_MIN:  continue
            if _edge_density(edges,x,y,r) < ED_MIN:   continue
            if _cu(hsv,x,y,r)             < CU_MIN:   continue
            if _brightness(hsv,x,y,r)     < BR_MIN:   continue
            # Gradient filter (skip when gc_min == 0.0)
            if gc_min > 0.0 and _gradient_consistency(gray,x,y,r) < gc_min:
                continue
            valid.append((x,y,r))
        det = len(valid)
        per_image[fname]  = det
        total_err         += abs(det - truth)
        total_over        += max(0, det - truth)
    return total_err, total_over, per_image

def main():

    print("Coin Detection - Gradient Consistency Threshold Search")
    print("=" * 62)
    print(f"Fixed: param2={PARAM2}  arc_min={ARC_MIN}  "
          f"ed_min={ED_MIN}  cu_min={CU_MIN}")
    print(f"Searching: gc_min in {GC_VALUES}\n")

    if not IMAGES_DIR.exists():
        print(f"[ERROR] IMAGES_DIR not found: {IMAGES_DIR.resolve()}")
        sys.exit(1)

    print(f"Caching {len(GROUND_TRUTH)} images...")
    t0    = time.time()
    cache = build_cache(IMAGES_DIR, GROUND_TRUTH)
    print(f"\n  {len(cache)} images ready in {time.time()-t0:.1f}s\n")

    results = []
    t0 = time.time()
    for gc_min in GC_VALUES:
        label = f"gc_min={gc_min:.2f}" if gc_min > 0 else "gc_min=OFF"
        err, over, pi = evaluate(gc_min, cache)
        results.append((err, over, gc_min, pi))
        elapsed = time.time() - t0
        print(f"  {label:14s}  error={err:4d}  overcounts={over:4d}  "
              f"({elapsed:.0f}s elapsed)", flush=True)

    results.sort(key=lambda r: (r[0], r[1]))

    print(f"\n{'='*62}")
    print("RESULTS  (sorted by total error)")
    print(f"{'='*62}")
    print(f"  {'gc_min':>8}  {'TotalErr':>8}  {'OverCnt':>7}  {'UnderCnt':>8}")
    print(f"  {'-'*45}")
    for err, over, gc, _ in results:
        under  = err - over
        label  = f"{gc:.2f}" if gc > 0 else "OFF"
        marker = " <- BEST" if (err,over,gc) == results[0][:3] else ""
        print(f"  {label:>8}  {err:8d}  {over:7d}  {under:8d}{marker}")

    best_err, best_over, best_gc, best_pi = results[0]
    best_under = best_err - best_over

    print(f"\n{'='*62}")
    print(f"BEST: gc_min={best_gc}  error={best_err}  "
          f"over={best_over}  under={best_under}")
    print(f"{'='*62}")

    n_imgs = n_exact = 0
    worst = []
    for fname, truth in GROUND_TRUTH.items():
        if fname not in best_pi: continue
        det  = best_pi[fname]
        diff = det - truth
        flag = "[checkmark]" if diff == 0 else ("[OVER]" if diff > 0 else "[UNDER]")
        print(f"  {flag} {fname:<12}  det={det}  truth={truth}  {diff:+d}")
        n_imgs += 1
        if diff == 0: n_exact += 1
        else: worst.append((fname, det, truth, diff))

    acc = 100*n_exact/n_imgs if n_imgs else 0
    print(f"\n  Exact: {n_exact}/{n_imgs}  ({acc:.0f}%)")

    if worst:
        print(f"\n  Remaining misses:")
        for fname, det, truth, diff in sorted(worst, key=lambda x: -abs(x[3])):
            print(f"    {fname}  det={det}  truth={truth}  {diff:+d}")

    print(f"\n{'='*62}")
    print("PASTE INTO detect.py -> filter_false_positives()")
    print("=" * 62)
    print(f"""
gradient_consistency_min = {best_gc}

# Full threshold set:
arc_coverage_min         = {ARC_MIN}
edge_density_min         = {ED_MIN}
color_uniformity_min     = {CU_MIN}
brightness_min           = {BR_MIN}
gradient_consistency_min = {best_gc}

# HoughCircles:
param1 = {PARAM1},  param2 = {PARAM2},  minDist = {MIN_DIST}
""")


if __name__ == "__main__":
